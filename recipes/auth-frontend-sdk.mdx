    ```javascript JavaScript
    import { Auth } from '@connect-xyz/auth-js';

    // Function to fetch JWT token from backend
    async function fetchJwtToken() {
      try {
        const response = await fetch('https://api.yourcompany.com/api/jwt', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
          },
        });

        const data = await response.json();
        return data.token; // Assuming the API returns { token: "jwt_string" }
      } catch (error) {
        console.error('Error fetching JWT:', error);
        throw error;
      }
    }

    // Initialize Auth widget with fetched JWT token
    async function initializeAuth() {
      // Fetch the JWT token from backend
      const jwtToken = await fetchJwtToken();

      // Create an Auth instance with configuration
      const auth = new Auth({
        jwt: jwtToken,
        env: 'production',
        theme: 'dark', // options: 'dark','light','auto'
        onError: ({ error, reason }) => {
          console.error('Auth error:', error, 'Reason:', reason);
        },
        onClose: () => {
          console.log('Auth widget closed');
        },
        onDeposit: ({ data }) => {
          console.log('Deposit initiated:', data);
        },
        onEvent: ({ type, data }) => {
          console.log('Event received:', type, data);
        },
      });

      // Render the widget to a container
      const container = document.getElementById('auth-container');
      await auth.render(container);
  
      // You can also destroy the widget at any time by using
      // auth.destroy()
    }

    // Call the initialization function
    initializeAuth();
    ```
