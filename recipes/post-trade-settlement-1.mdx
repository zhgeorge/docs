## Full example

```python Python
import hmac
import hashlib
import json
from base64 import b64decode, b64encode
from datetime import datetime
from typing import Any, Dict, Optional
from urllib.parse import urljoin, urlencode
import time

import requests
from logging import getLogger

BASE_URL = 'BASE_URL'
API_PUBLIC_KEY = "API_PUBLIC_KEY"
API_PRIVATE_KEY = "API_PRIVATE_KEY"
PASSPHRASE = "PASSPHRASE"

def sign_request(api_private_key: str, method: str,resource_path: str, json_body: str, timestamp: str) -> bytes:
    msg = bytes(timestamp + method + resource_path + json_body,encoding='utf-8')
    hm = hmac.new(key=b64decode(api_private_key),msg=msg,digestmod=hashlib.sha256)

    return b64encode(hm.digest()).decode()

def build_headers() -> Dict[str, Any]:
    timestamp = str(int(datetime.now().timestamp()))

    return {
        'X-SCX-API-KEY': API_PUBLIC_KEY,
        'X-SCX-SIGNED': '<to be filled>', # PLEASE NOTE: will be auto filled below
        'X-SCX-TIMESTAMP': timestamp,
        'X-SCX-PASSPHRASE': PASSPHRASE
    }

def send_signed_http_request(method: str, resource_path: str, body: Optional[Dict[str, str]] = None) -> requests.Response:
    if body is None:
        body = {}
        
    headers = build_headers()
    json_body = json.dumps(body, separators=(',', ':'))
    signature = sign_request(
        API_PRIVATE_KEY,
        method,
        resource_path,
        json_body,
        headers['X-SCX-TIMESTAMP']
    )
    headers['X-SCX-SIGNED'] = signature

    absolute_url = urljoin(BASE_URL, resource_path)

    request_args = {
        'method': method,
        'url': absolute_url,
        'headers': headers
    }
    if body:
        request_args['data'] = json_body
        headers['Content-Type'] = 'application/json'

    response = requests.request(**request_args)

    return json.loads(response.text)
```
